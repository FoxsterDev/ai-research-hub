    (() => {
      const roleButtons = Array.from(document.querySelectorAll("[data-role]"));
      const rolePanels = Array.from(document.querySelectorAll("[data-role-panel]"));
      const roleLinks = Array.from(document.querySelectorAll("[data-role-target]"));

      if (!roleButtons.length || !rolePanels.length) {
        return;
      }

      const setRole = (role) => {
        roleButtons.forEach((button) => {
          const isActive = button.dataset.role === role;
          button.setAttribute("aria-pressed", isActive ? "true" : "false");
        });

        rolePanels.forEach((panel) => {
          panel.classList.toggle("active", panel.dataset.rolePanel === role);
        });

        try {
          localStorage.setItem("airroot-role", role);
        } catch (error) {
        }
      };

      roleButtons.forEach((button) => {
        button.addEventListener("click", () => setRole(button.dataset.role));
      });

      roleLinks.forEach((link) => {
        link.addEventListener("click", () => setRole(link.dataset.roleTarget));
      });

      let initialRole = "founder";

      try {
        initialRole = localStorage.getItem("airroot-role") || initialRole;
      } catch (error) {
      }

      if (!rolePanels.some((panel) => panel.dataset.rolePanel === initialRole)) {
        initialRole = "founder";
      }

      setRole(initialRole);
    })();
  </script>
